    Read x-portkey-cache-status from a raw response obtained via
    `client.chat.completions.with_raw_response.create(...)`.

_rails: LLMRails | None = None


def initialize_rails() -> None:
    """
    Build the NeMo LLMRails singleton at app startup.
    Uses DeepSeek for intent classification at the gate.
    """
    global _rails

    guard_llm = ChatOpenAI(
        api_key=settings.deepseek_api_key,
        base_url=settings.deepseek_base_url,
        model=settings.deepseek_model,
        temperature=0,
    )

    config = RailsConfig.from_content(
        colang_content=COLANG_CONTENT,
        yaml_content=YAML_CONTENT,
    )

    _rails = LLMRails(config, llm=guard_llm)
    log.info(f"🛡️ NeMo Guardrails initialised ({settings.deepseek_model}).")
    
    


def guard(message: str) -> tuple[bool, str | None]:
    """
    Run a user message through the NeMo rails gate.

    Returns:
        (True,  rail_response) — a rail fired; return this response immediately,
                                skip the RAG pipeline entirely.
        (False, None)          — message is clean; proceed to LangGraph.
    """
    if _rails is None:
        log.warning("⚠️ Guardrails not initialised — skipping gate.")
        return False, None

    with log.span("🛡️ Guardrails Check"):
        result = _rails.generate(messages=[{"role": "user", "content": message}])

        # NeMo returns {'role': 'assistant', 'content': '...'} — extract text
        content = result.get("content", "") if isinstance(result, dict) else str(result)

        fired = any(indicator in content for indicator in RAIL_INDICATORS)

        if fired:
            log.info(f"🛡️ Guardrails fired | query='{message[:80]}'")
            return True, content

        log.info("✅ Guardrails passed.")
        return False, None

from typing import List
from app.logger import log

def chunk_text(text: str, chunk_size: int = 1500) -> List[str]:
    """
    Simple semantic-ish chunker that splits by paragraphs.
    Ensures chunks do not exceed the specified size.
    """
    with log.span("✂️ Text Chunking", text_length=len(text)):
        if not text.strip(): 
            return []
            
        paragraphs = text.split("\n\n")
        chunks = []
        current_chunk = ""
        
        for p in paragraphs:
            if len(current_chunk) + len(p) < chunk_size:
                current_chunk += p + "\n\n"
            else:
                if current_chunk.strip():
                    chunks.append(current_chunk.strip())
                current_chunk = p + "\n\n"
        
        if current_chunk.strip():
            chunks.append(current_chunk.strip())
            
        valid_chunks = [c for c in chunks if c.strip()]
        log.info(f"✅ Generated {len(valid_chunks)} chunks")
        return valid_chunks

def get_loader(ext: str) -> Callable[[str], str] | None:
    """Return the parser function for an extension, or None if unsupported."""
    entry = LOADERS.get(ext)
    if entry is None:
        return None
    module_path, func_name = entry
    module = importlib.import_module(module_path)  # raises ImportError if dep missing
    return getattr(module, func_name)


def save_processed_locally(data: dict, source_type: str, filename: str) -> str:
    """Save parsed chunk metadata as JSON in processed_data/<source_type>/."""
    folder = os.path.join(PROCESSED_DATA_DIR, source_type)
    os.makedirs(folder, exist_ok=True)
    dest = os.path.join(folder, f"{filename}.json")
    with open(dest, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)
    return dest


def ensure_index(wipe: bool = False):
    """Create the Pinecone index if needed (optionally deleting it first) and return it."""
    name = settings.pinecone_index_name

    if wipe and pc.has_index(name):
        with log.span("Wiping Index"):
            pc.delete_index(name)
            log.info("Index '{name}' deleted.", name=name)

    if not pc.has_index(name):
        dim = get_embedding_dim()  # resolved at runtime from the embedding model
        pc.create_index(
            name=name,
            dimension=dim,
            metric="cosine",
            spec=ServerlessSpec(cloud="aws", region=settings.pinecone_region),
        )
        log.info("Created index '{name}' ({dim}-dim, cosine).", name=name, dim=dim)

    return pc.Index(name)


def upsert_in_batches(index, vectors: list[dict]) -> None:
    for i in range(0, len(vectors), UPSERT_BATCH_SIZE):
        index.upsert(vectors=vectors[i : i + UPSERT_BATCH_SIZE])

MAX_CHUNK_CHARS = 6000
CHUNK_OVERLAP_CHARS = 200


def enforce_chunk_size(chunks: list[str]) -> list[str]:
    """Split oversized chunks before embedding and Pinecone indexing."""
    safe_chunks = []

    for chunk in chunks:
        if len(chunk.encode("utf-8")) <= 30 * 1024:
            safe_chunks.append(chunk)
            continue

        start = 0

        while start < len(chunk):
            end = min(start + MAX_CHUNK_CHARS, len(chunk))

            # Prefer breaking at a word boundary.
            if end < len(chunk):
                boundary = chunk.rfind(" ", start, end)
                if boundary > start:
                    end = boundary

            piece = chunk[start:end].strip()

            if piece:
                # Verify the actual UTF-8 metadata size.
                if len(piece.encode("utf-8")) > 30 * 1024:
                    raise ValueError("Chunk exceeds the safe metadata limit.")

                safe_chunks.append(piece)

            if end >= len(chunk):
                break

            start = max(end - CHUNK_OVERLAP_CHARS, start + 1)

    return safe_chunks
def process_file(index, file_path: str, filename: str, source_type: str):
    """Parse → chunk → save locally → embed → index in Pinecone."""
    with log.span("Processing File", file=filename, source=source_type):
        try:
            # 1. Extract text based on file extension
            ext = os.path.splitext(filename)[1].lstrip(".").lower()
            loader = get_loader(ext)
            if loader is None:
                log.warn("Skipping unsupported file type: {file}", file=filename)
                return

            full_text = loader(file_path)

            if not full_text or not full_text.strip():
                log.warn("No text extracted from {file}, skipping.", file=filename)
                return

            # 2. Chunk text
            # 2. Chunk text and enforce Pinecone metadata limits
            chunks = chunk_text(full_text)

            if not chunks:
                log.warn("No chunks generated for {file}.", file=filename)
                return

            original_count = len(chunks)
            chunks = enforce_chunk_size(chunks)

            log.info(
                "Chunked {file}: {original} original chunks → {final} safe chunks.",
                file=filename,
                original=original_count,
                final=len(chunks),
                )

            # 3. Save processed data locally
            processed_data = {
                "filename": filename,
                "source_type": source_type,
                "chunks": chunks,
            }
            local_path = save_processed_locally(processed_data, source_type, filename)
            log.info("Saved processed data → {path}", path=local_path)

            # 4. Embed and index in Pinecone
            with log.span("Vectorizing & Indexing"):
                embeddings = embed_texts(chunks)
                vectors = [
                    {
                        # same file + chunk number always gives the same ID,
                        # so re-running ingestion overwrites instead of duplicating
                        "id": str(uuid.uuid5(uuid.NAMESPACE_URL, f"{source_type}/{filename}/{i}")),
                        "values": vector,
                        "metadata": {
                            "text": chunk,
                            "source": filename,
                            "source_type": source_type,
                        },
                    }
                    for i, (chunk, vector) in enumerate(zip(chunks, embeddings))
                ]
                upsert_in_batches(index, vectors)
                log.info(
                    "Indexed {n} vectors to Pinecone from {file}.",
                    n=len(vectors),
                    file=filename,
                )

        except Exception:
            # Also catches ImportError from a missing loader dependency,
            # so one bad file type never stops the rest of the run.
            log.exception("Failed to process {file}", file=filename)


def process_directory(index, dir_path: str, source_type: str):
    """Process every file in a directory."""
    with log.span("Scanning Directory", path=dir_path, source=source_type):
        files = [
            f for f in os.listdir(dir_path)
            if os.path.isfile(os.path.join(dir_path, f))
        ]
        log.info("Found {n} files in {path}.", n=len(files), path=dir_path)
        for filename in files:
            process_file(index, os.path.join(dir_path, filename), filename, source_type)


def run_universal_ingestion(
    base_dir: str, explicit_source_type: str = None, wipe: bool = False
):
    """
    Scan base_dir, map sub-folders to source types, and ingest all documents.
    Pass --wipe to delete and recreate the Pinecone index before ingestion.
    """
    with log.span("Universal Ingestion Started", base_directory=base_dir):
        index = ensure_index(wipe=wipe)
        log.info(
            "Processing directory '{path}' as '{source}'.",
            path=base_dir,
            source=explicit_source_type,
        )
        process_directory(index, base_dir, explicit_source_type)
        

if __name__ == "__main__":
    wipe_requested = "--wipe" in sys.argv
    clean_args = [a for a in sys.argv if a != "--wipe"]

    target_dir = clean_args[1] if len(clean_args) > 1 else "app/data"
    explicit_type = clean_args[2] if len(clean_args) > 2 else "general"

    if not os.path.isdir(target_dir):
        print(f"Error: directory '{target_dir}' does not exist.")
        sys.exit(1)

    run_universal_ingestion(
        target_dir,
        explicit_source_type=explicit_type,
        wipe=wipe_requested,
    )

    log.info("Ingestion job completed.")


embedding.py
import time

from pinecone import Pinecone

from app.config import settings
from app.logger import log

BATCH_SIZE = 96   # Pinecone inference limit per embed request
MAX_RETRIES = 4

_pc: Pinecone | None = None


# ── Client initialisation ──────────────────────────────────────────────────────

def _get_client() -> Pinecone:
    """Create the Pinecone client once per process, on first use."""
    global _pc
    if _pc is None:
        _pc = Pinecone(api_key=settings.pinecone_api_key)
        log.info("Pinecone embeddings ready ({model}, {dim}-dim).",
                 model=settings.embedding_model, dim=settings.embedding_dim)
    return _pc


# ── Public helpers ─────────────────────────────────────────────────────────────

def get_embedding_dim() -> int:
    """Vector dimension used when creating the Pinecone index."""
    return settings.embedding_dim


# ── Batch embedding with retry ─────────────────────────────────────────────────

def _embed_batch(batch: list[str], input_type: str) -> list[list[float]]:
    pc = _get_client()

    # Exponential backoff: 1 s → 2 s → 4 s (4 attempts total)
    for attempt in range(MAX_RETRIES):
        try:
            res = pc.inference.embed(
                model=settings.embedding_model,
                inputs=batch,
                parameters={"input_type": input_type, "truncate": "END"},
            )
            return [item["values"] for item in res]
        except Exception as e:
            err = str(e).lower()
            is_rate_limit = any(x in err for x in ("429", "rate", "quota", "too many"))
            if is_rate_limit and attempt < MAX_RETRIES - 1:
                wait = 2 ** attempt
                log.warn(
                    "Pinecone rate limit hit, retrying in {wait}s (attempt {n}/{total}).",
                    wait=wait, n=attempt + 1, total=MAX_RETRIES,
                )
                time.sleep(wait)
            else:
                log.exception("Pinecone embedding failed")
                raise

    raise RuntimeError(f"Pinecone rate limit persisted after {MAX_RETRIES} attempts.")


def _check_dim(vectors: list[list[float]]) -> None:
    # Catch a model/index mismatch early instead of failing at upsert time
    if vectors and len(vectors[0]) != settings.embedding_dim:
        raise ValueError(
            f"Embedding model returned {len(vectors[0])}-dim vectors but "
            f"settings.embedding_dim is {settings.embedding_dim}. "
            f"Fix embedding_dim in config to match '{settings.embedding_model}'."
        )


# ── Public API ─────────────────────────────────────────────────────────────────

def embed_query(query: str) -> list[float]:
    """Embed a single search query."""
    vectors = _embed_batch([query], input_type="query")
    _check_dim(vectors)
    return vectors[0]


def embed_texts(texts: list[str]) -> list[list[float]]:
    """Embed documents/chunks for indexing."""
    all_embeddings: list[list[float]] = []
    for i in range(0, len(texts), BATCH_SIZE):
        batch = texts[i : i + BATCH_SIZE]
        with log.span("Embed batch", model=settings.embedding_model, start=i, size=len(batch)):
            all_embeddings.extend(_embed_batch(batch, input_type="passage"))
    _check_dim(all_embeddings)
    return all_embeddings

from app.logger import log
from pinecone import Pinecone
from app.config import settings
from app.services.retrieval.embedding import embed_query


# Initialize Pinecone client
pc = Pinecone(api_key=settings.pinecone_api_key)

# Connect to the existing Pinecone index
index = pc.Index(settings.pinecone_index_name)


def search_enterprise_knowledge(query: str, limit: int = 8):
    """
    Performs semantic similarity search in the enterprise
    knowledge base using Pinecone.

    Returns:
        List of dictionaries containing content, source, and score.
    """
    try:
        query_vector = embed_query(query)

        response = index.query(
            vector=query_vector,
            top_k=limit,
            include_metadata=True,
            include_values=False,
        )

        results = []

        for match in response.get("matches", []):
            metadata = match.get("metadata") or {}

            results.append({
                "content": metadata.get("text", ""),
                "source": metadata.get("source", "Unknown"),
                "score": match.get("score", 0.0),
            })

        return results

    except Exception:
        log.exception("Pinecone enterprise knowledge search failed")
        return []

import time

# Lazy initialization - Ranker is loaded on first use to ensure logfire.configure() has run
_ranker = None


def _get_ranker() -> Ranker:
    """
    Initializes the FlashRank engine lazily. 
    FlashRank uses a local ONNX model (ms-marco-MiniLM-L-6-v2) for ultra-fast reranking.
    """
    global _ranker
    if _ranker is None:
        log.info("🧠 Initializing FlashRank Model (TinyBERT) locally...")
        try:
            # We use a specific cache directory to avoid permission issues in production
            _ranker = Ranker(cache_dir="/tmp/flashrank")
        except Exception:
            _ranker = Ranker()
    return _ranker



def rerank_documents(query: str, documents: list[str], top_n: int = 5) -> list[str]:
    """
    Refines retrieval results by re-scoring documents against the query semantically.
    
    Why FlashRank? 
    Standard vector search (Cosine Similarity) is fast but mathematically "fuzzy."
    FlashRank uses a Cross-Encoder approach which is much more precise but usually slow.
    FlashRank solves this by using highly optimized, quantized ONNX models locally.
    """
    if not documents:
        return []

    start_time = time.time()
    log.info(f"📡 [Reranker] Sending {len(documents)} docs to FlashRank Cross-Encoder...")

    try:
        ranker = _get_ranker()
        
        # FlashRank expects a list of dictionaries with 'id' and 'text'
        passages = [
            {"id": i, "text": doc}
            for i, doc in enumerate(documents)
        ]

        request = RerankRequest(query=query, passages=passages)
        results = ranker.rerank(request)
        
        # Results are returned sorted by highest semantic score first
        reranked_docs = []
        for res in results[:top_n]:
            reranked_docs.append(res['text'])

        duration = time.time() - start_time
        top_score = results[0]['score'] if results else 'N/A'
        log.info(f"✅ [Reranker] Done in {duration:.2f}s. Top semantic score: {top_score}")
        
        return reranked_docs

    except Exception as e:
        log.error(f"❌ [Reranker] Semantic Reranking Failed: {e}")
        # Fallback to the original Qdrant order to ensure the user still gets an answer
        return documents[:top_n]

from typing import TypedDict, List, Annotated
import operator


class AgentState(TypedDict):
    # Using Annotated with operator.add ensures that messages 
    # are appended to the history rather than replaced.
    messages: Annotated[List[dict], operator.add]

    documents: List[str]

from app.agents.state import AgentState
from app.gateway import get_langchain_llm
from app.logger import log

# Portkey-backed LLM: fallback + cache + retry — same .invoke() interface as ChatGroq
llm = get_langchain_llm(feature="planner")

def planner_node(state: AgentState):
    """
    The Planner determines if a search is needed based on the ENTIRE conversation.
    """
    # Get the conversation history (excluding the latest message)
    history = ""
    for msg in state["messages"][:-1]:
        role = "User" if msg["role"] == "user" else "Assistant"
        history += f"{role}: {msg['content']}\n"
    
    user_message = state["messages"][-1]["content"] if state["messages"] else ""
    
    prompt = f"""
You are the Planner for a Goa travel guide assistant.
Analyze the conversation history and the latest user message.

CONVERSATION HISTORY:
{history}

LATEST MESSAGE:
"{user_message}"

Choose exactly ONE route:
1. CONVERSATIONAL - greetings, thanks, or questions answerable using ONLY the conversation history (e.g. "what is my name", "what did you suggest earlier").
2. DOCS - stable Goa knowledge in our guide: beaches, forts, churches, temples, food and dishes, culture, history, festivals (general), itineraries, neighborhoods, travel tips.
3. WEB - anything time-sensitive or changeable: today's weather, current events or dates, prices, opening hours, ferry/bus/train timings, new restaurants or clubs, news, safety alerts, flight info.

Output format (one line only):
CONVERSATIONAL
DOCS: <refined search query>
WEB: <refined search query>
"""
    
    with log.span("🧠 Planner Decision"):
        decision = llm.invoke(prompt).content.strip()
        log.info(f"Intent identified: {decision}")

    upper = decision.upper()
    
    if upper.startswith("CONVERSATIONAL"):
        return {
            "route": "conversational",
            "current_query": "",
            "status": "Handling conversationally (using memory)...",
            "plan": ["Intent: Conversational/Memory", "Retrieval: Skipped"],
        }

    if upper.startswith("WEB"):
        query = decision.split(":", 1)[-1].strip() or user_message
        return {
            "route": "web",
            "current_query": query,
            "status": f"Searching the web for: {query}",
            "plan": ["Intent: Needs fresh info", f"Tavily Query: {query}"],
        }
    
    query = decision.split(":", 1)[-1].strip() if ":" in decision else decision
    return {
        "route": "docs",
        "current_query": query or user_message,
        "status": f"Technical research needed. Searching docs for: {query}",
        "plan": ["Intent: Technical", f"Search Term: {query}"],
    }

from app.agents.state import AgentState
from app.gateway import portkey_client, extract_cache_status
from app.logger import log
from app.config import settings

def generate_node(state: AgentState):
    """
    Synthesizes a response using both Documentation Context AND Conversation History.
    Uses the native Portkey client (not LangChain) so we can read the
    x-portkey-cache-status response header and surface Cache: Hit in the UI.
    """
    query = state["current_query"]

    history_str = ""
    for msg in state["messages"][:-1]:
        role = "User" if msg["role"] == "user" else "Assistant"
        history_str += f"{role}: {msg['content']}\n"

    user_msg = state["messages"][-1]["content"] if state["messages"] else ""

    if query == "CONVERSATIONAL":
        log.info("Generating conversational response using memory.")
        prompt = f"""
        You are a friendly and helpful Enterprise AI Assistant.
        Answer the user's latest message using the CONVERSATION HISTORY below.

        CONVERSATION HISTORY:
        {history_str}

        LATEST MESSAGE:
        "{user_msg}"
        """
    else:
        log.info("Generating technical RAG response.")
        max_context_chars = 25000
        full_context = ""

        for doc in state["documents"]:
            if len(full_context) + len(doc) < max_context_chars:
                full_context += doc + "\n\n"
            else:
                log.warning("Context truncated to fit Groq TPM limits.")
                break

        prompt = f"""
        You are a Senior Technical Architect.
        Answer the question using the TECHNICAL CONTEXT provided.

        TECHNICAL CONTEXT:
        {full_context}

        CONVERSATION HISTORY:
        {history_str}

        USER QUESTION:
        "{user_msg}"
        """

    with log.span("✍️ LLM Synthesis"):
        try:
            response = portkey_client.chat.completions.create(
                model=f"@{settings.GROQ_SLUG}/{settings.deepseek_model}",
                messages=[{"role": "user", "content": prompt}],
                temperature=0.1
            )
            content = response.choices[0].message.content
            cache_status = extract_cache_status(response)
            is_cache_hit = cache_status == "HIT"

            if is_cache_hit:
                log.info("⚡ Gateway Cache Hit — response served from Portkey cache.")
                plan_update = state["plan"] + ["Cache: Hit ⚡"]
                status = "Cache hit — instant response."
            else:
                log.info("✅ Response synthesised via LLM.")
                plan_update = state["plan"]
                status = "Response generated."

            return {
                "final_answer": content,
                "status": status,
                "plan": plan_update,
                "messages": [{"role": "assistant", "content": content}]
            }

        except Exception as e:
            log.error(f"LLM Generation failed: {e}")
            raise e

from app.logger import log
from app.agents.state import AgentState
from app.services.retrieval.pinecone_service import search_enterprise_knowledge
from app.services.retrieval.ranking_service import rerank_documents

def retrieve_node(state: AgentState):
    """
    Performs vector search and semantic reranking for technical queries.
    """
    query = state["current_query"]
    
    
    # Standard Retrieval Logic
    with log.span("🔍 Knowledge Retrieval"):
        log.info(f"Searching Qdrant for: {query}")
        raw_results = search_enterprise_knowledge(query, limit=15)
        log.info(f"Retrieved {len(raw_results)} candidates from Vector DB")
        
        doc_contents = [doc['content'] for doc in raw_results]
        
        with log.span("⚖️ Semantic Reranking"):
            reranked_contents = rerank_documents(query, doc_contents, top_n=5)
            log.info("Reranking complete. Kept top 5 most relevant chunks.")
            
        formatted_docs = [f"CONTENT: {doc}" for doc in reranked_contents]
    
    return {
        "documents": formatted_docs,
        "status": f"Found technical context.",
        "plan": state["plan"] + ["Context Retrieved"]
    }

# app/agents/nodes/web_search.py
import os
from tavily import TavilyClient
from app.agents.state import AgentState
from app.logger import log

tavily = TavilyClient(api_key=os.environ["TAVILY_API_KEY"])

def web_search_node(state: AgentState):
    query = state["current_query"]

    with log.span("🌐 Tavily Web Search"):
        try:
            response = tavily.search(
                query=query,
                search_depth="basic",     # "advanced" = better quality, 2x credits
                max_results=5,
                include_answer=False,
            )
            results = response.get("results", [])
        except Exception as e:
            log.error(f"Tavily search failed: {e}")
            results = []

        log.info(f"Tavily returned {len(results)} results")

    # Format into the same shape the responder already consumes (List[str])
    documents = [
        f"[Source: {r['title']} - {r['url']}]\n{r['content']}"
        for r in results
    ]
    sources = [r["url"] for r in results]

    return {
        "documents": documents,
        "sources": sources,
        "status": f"Found {len(results)} web results",
    }
